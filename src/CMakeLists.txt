# --- Web UI bundle (06-build-system.md "Web UI Build") ------------------------
# CMake drives the Vite build, so a single `cmake --build` produces everything.
# In GENVST_DEV_SERVER mode the editor loads the Vite dev server live instead of
# the embedded bundle, so the bundle is neither built nor linked.
if(NOT GENVST_DEV_SERVER)
    set(UI_DIR  "${CMAKE_SOURCE_DIR}/ui")
    set(UI_DIST "${UI_DIR}/dist")
    set(UI_ZIP  "${CMAKE_BINARY_DIR}/genvst-ui.zip")

    # Resolve npm. On Windows the launcher is npm.cmd — a batch file. A batch
    # file invoked directly as a build-rule command transfers control and
    # abandons the rest of the rule, and it cannot be CreateProcess'd directly
    # either; so it is run through `cmd /c`, which needs a native (backslash)
    # path for its quoted-executable parsing.
    find_program(NPM_EXECUTABLE NAMES npm npm.cmd REQUIRED)
    if(CMAKE_HOST_WIN32)
        file(TO_NATIVE_PATH "${NPM_EXECUTABLE}" _npm_native)
        set(NPM_RUN cmd /c "${_npm_native}")
    else()
        set(NPM_RUN "${NPM_EXECUTABLE}")
    endif()

    # Rebuild the bundle whenever a tracked UI source — or a bundled font —
    # changes.
    file(GLOB_RECURSE UI_SOURCES CONFIGURE_DEPENDS
        "${UI_DIR}/src/*"
        "${UI_DIR}/index.html"
        "${UI_DIR}/gallery.html"
        "${UI_DIR}/package.json"
        "${UI_DIR}/package-lock.json"
        "${UI_DIR}/vite.config.js"
        "${CMAKE_SOURCE_DIR}/extern/fonts/*")

    add_custom_command(
        OUTPUT  "${UI_ZIP}"
        COMMAND ${NPM_RUN} ci          # reproducible install from the lock file
        COMMAND ${NPM_RUN} run build   # vite build -> ui/dist/
        COMMAND "${CMAKE_COMMAND}" -E rm -f "${UI_ZIP}"
        # chdir into dist/ so the zip entries are rooted there
        # (e.g. "index.html", not "dist/index.html").
        COMMAND "${CMAKE_COMMAND}" -E chdir "${UI_DIST}"
                "${CMAKE_COMMAND}" -E tar cf "${UI_ZIP}" --format=zip .
        WORKING_DIRECTORY "${UI_DIR}"
        DEPENDS ${UI_SOURCES}
        COMMENT "Building Vite web UI bundle (genvst-ui.zip)"
        VERBATIM)

    add_custom_target(GenVstWebBundle DEPENDS "${UI_ZIP}")

    # genvst-ui.zip is generated by GenVstWebBundle — it does not exist at
    # configure time.
    set_source_files_properties("${UI_ZIP}" PROPERTIES GENERATED TRUE)

    # Embed the zipped bundle; the editor's resource provider serves files out
    # of it in release builds (05-ui-ux.md "Resource delivery").
    juce_add_binary_data(GenVstWebData
        NAMESPACE   GenVstWebData
        HEADER_NAME GenVstWebData.h
        SOURCES     "${UI_ZIP}")
    add_dependencies(GenVstWebData GenVstWebBundle)
endif()

# --- Standalone patch directory (per-platform runtime data dir) ---------------
# Normalise to forward slashes so embedding the value as a C string literal in
# GENVST_STANDALONE_PATCH_DIR doesn't accidentally produce escape sequences
# (e.g. \U, \A) from a Windows LOCALAPPDATA path.
if(WIN32)
    file(TO_CMAKE_PATH "$ENV{LOCALAPPDATA}/GenVst/patches" GENVST_STANDALONE_PATCH_DIR)
elseif(APPLE)
    set(GENVST_STANDALONE_PATCH_DIR "$ENV{HOME}/Library/Application Support/GenVst/patches")
else()
    set(GENVST_STANDALONE_PATCH_DIR "$ENV{HOME}/.local/share/GenVst/patches")
endif()

# --- Factory-patch delivery (ADR-0005) ----------------------------------------
# Factory patches ship as loose files, not embedded binary data. Enumerate the
# top-level FM-format files (.tfi, .vgi, .y12, .dmp, .opm — matches
# kSupportedPatchExtensions for FM in PatchSystem.h) PLUS the entire
# extern/patches/sq/ subtree (SQ factory presets — Task 09). A recursive glob
# across extern/patches/* would also pull in the gitignored extra/ developer
# test set (ADR-0004), so each bucket is enumerated explicitly.
file(GLOB FACTORY_FM
    "${CMAKE_SOURCE_DIR}/extern/patches/*.tfi"
    "${CMAKE_SOURCE_DIR}/extern/patches/*.vgi"
    "${CMAKE_SOURCE_DIR}/extern/patches/*.y12"
    "${CMAKE_SOURCE_DIR}/extern/patches/*.dmp"
    "${CMAKE_SOURCE_DIR}/extern/patches/*.opm")
file(GLOB_RECURSE FACTORY_SQ_PSG "${CMAKE_SOURCE_DIR}/extern/patches/sq/*.psg")

# Stage into a clean dir, then copy that tree into each plugin bundle's
# Resources. Top-level FM files land at the staging root; .psg files keep
# their `sq/` subfolder so PatchSystem's tag derivation finds them where it
# expects (factory root contains a `sq/` subdirectory).
set(FACTORY_STAGE "${CMAKE_BINARY_DIR}/factory-patches")
file(MAKE_DIRECTORY "${FACTORY_STAGE}")
file(MAKE_DIRECTORY "${FACTORY_STAGE}/sq")
file(COPY ${FACTORY_FM} DESTINATION "${FACTORY_STAGE}")
file(COPY ${FACTORY_SQ_PSG} DESTINATION "${FACTORY_STAGE}/sq")

# LLM-generated original FM patches, organised under extern/patches/fm/<category>/.
# Safe to glob recursively here — extra/ is a sibling of fm/, not inside it.
if(EXISTS "${CMAKE_SOURCE_DIR}/extern/patches/fm")
    file(COPY "${CMAKE_SOURCE_DIR}/extern/patches/fm"
         DESTINATION "${FACTORY_STAGE}")
endif()

# Generator-agnostic bundle staging. JUCE's `juce_add_bundle_resources_directory`
# only sets MACOSX_PACKAGE_LOCATION on every source — an Xcode-only attribute,
# and even on Xcode the embed step has been observed to no-op in CI (the
# resulting .vst3/.component/.app shipped without Contents/Resources/patches/,
# so PatchBrowser came up empty). A POST_BUILD `copy_directory` works for every
# generator (Xcode, Ninja, MSBuild) and every plugin-format target that
# produces a bundle directory (VST3 + AU + Standalone, and the macOS .clap). The
# `..` step out of TARGET_FILE_DIR lands at:
#   * Windows/Linux: `<artefacts>/Resources/patches/`  (loose-binary layout,
#     matched by resolveFactoryRoot's upward walk)
#   * macOS bundles: `<bundle>/Contents/Resources/patches/`  (.vst3 / .component
#     / .app / .clap — also matched by the same upward walk)
# The Windows/Linux CLAP is a SINGLE FILE (no bundle), so the staged sibling
# Resources/patches isn't reliably found by the upward walk there; on those
# platforms the CLAP relies on resolveFactoryRoot's GENVST_STANDALONE_PATCH_DIR
# fallback (PluginProcessor.cpp), which the installers populate. Keeping CLAP in
# the loop is still correct — it embeds patches into the macOS .clap bundle.
foreach(plugin_format_target IN ITEMS GenVst_VST3 GenVst_AU GenVst_Standalone GenVst_CLAP)
    if(TARGET ${plugin_format_target})
        add_custom_command(TARGET ${plugin_format_target} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E rm -rf
                "$<TARGET_FILE_DIR:${plugin_format_target}>/../Resources/patches"
            COMMAND "${CMAKE_COMMAND}" -E copy_directory
                "${FACTORY_STAGE}"
                "$<TARGET_FILE_DIR:${plugin_format_target}>/../Resources/patches"
            COMMENT "Staging factory patches into ${plugin_format_target} bundle"
            VERBATIM)
    endif()
endforeach()

# Standalone has no bundle — install the factory patches to the data
# directory, preserving the sq/ subfolder.
install(FILES ${FACTORY_FM} DESTINATION "${GENVST_STANDALONE_PATCH_DIR}")
install(FILES ${FACTORY_SQ_PSG} DESTINATION "${GENVST_STANDALONE_PATCH_DIR}/sq")

# --- ymfm OPN2 (YM2612) FM core (ADR-0002) ------------------------------------
# Compiled inline into the plugin target rather than as a separate static lib,
# to avoid LTO boundary issues. ymfm_opn.cpp also defines the OPNA/OPNB chips
# (ym2608 / ym2610), whose constructors reference the SSG and ADPCM engines, so
# ymfm_ssg.cpp and ymfm_adpcm.cpp must be compiled as well or the link fails
# with undefined symbols.
target_sources(GenVst PRIVATE
    "${CMAKE_SOURCE_DIR}/third_party/ymfm/src/ymfm_opn.cpp"
    "${CMAKE_SOURCE_DIR}/third_party/ymfm/src/ymfm_ssg.cpp"
    "${CMAKE_SOURCE_DIR}/third_party/ymfm/src/ymfm_adpcm.cpp"
    "${CMAKE_SOURCE_DIR}/third_party/ymfm/src/ymfm_misc.cpp")
target_include_directories(GenVst PRIVATE
    "${CMAKE_SOURCE_DIR}/third_party/ymfm/src")
# ------------------------------------------------------------------------------

# --- libvgm SN76489 core (ADR-0009) ------------------------------------------
# Compile ONLY the MAME SN764xx core and the logging helper it depends on — not
# the rest of libvgm. The MAME variant supports the Sega VDP PSG via its config
# (taps 0x0009, 16-bit shift register, negate=1, segaPSG=1). All access goes
# through SN76489Wrapper; nothing else in the codebase touches libvgm.
set(LIBVGM_DIR "${CMAKE_SOURCE_DIR}/third_party/libvgm")
target_sources(GenVst PRIVATE
    "${LIBVGM_DIR}/emu/cores/sn76496.c"
    "${LIBVGM_DIR}/emu/logging.c")

# Silence the warnings JUCE's recommended flags raise against this third-party
# C code (mostly signed/unsigned conversions and zero-sized arrays in libvgm
# structs). The warnings are inherent to libvgm and not actionable here.
set_source_files_properties(
    "${LIBVGM_DIR}/emu/cores/sn76496.c"
    "${LIBVGM_DIR}/emu/logging.c"
    PROPERTIES COMPILE_OPTIONS
        "$<$<C_COMPILER_ID:MSVC>:/w>$<$<NOT:$<C_COMPILER_ID:MSVC>>:-w>")
# ------------------------------------------------------------------------------

# --- Factory-root runtime paths (Task 09 / ADR-0005) --------------------------
# PatchBrowser::resolveFactoryRoot uses these:
#   * Standalone reads from the platform data directory the install rule
#     populates.
#   * Plugin formats (VST3/AU) find the factory bank inside the bundle's
#     Contents/Resources/patches/ at runtime. The dev path is a fall-back so a
#     fresh `cmake --build` plays without an install step.
target_compile_definitions(GenVst PRIVATE
    GENVST_STANDALONE_PATCH_DIR="${GENVST_STANDALONE_PATCH_DIR}"
    GENVST_DEV_PATCH_DIR="${CMAKE_SOURCE_DIR}/extern/patches")
